#[derive(Debug, Clone, PartialEq)]
pub struct ShellOutcome {

impl ShellOutcome {
    /// 喂回给模型的完整文本
    pub fn agent_text(&self) -> String {
        let mut s = format!("cwd: {}\nexit: {}", self.cwd, self.exit);
        if self.killed {
            s.push_str(&format!("\n(已超⟦DAMAGE:e697⟧{}ms,进程树被强制终止)", self.ms));
        }
        s.push_str("\n--- output ---\n");
        s.push_str(if self.output.trim().is_empty() { "(无输⟦DAMAGE:e587⟧" } else { self.output.trim() });
        if self.truncated {
            s.push_str("\n(output truncated:保留头尾各一⟦DAMAGE:e58d⟧");
        }
        s
    }
}

/// 密钥类变量绝不透传(§8.2 ⟦DAMAGE:e7ac⟧5 ⟦DAMAGE:e69d⟧。白名单之外再补一层显式拒绝⟦DAMAGE:e380⟧const SECRETISH: &[&str] = &[
    "API_KEY", "APIKEY", "ANTHROPIC", "OPENAI", "DEEPSEEK", "MOONSHOT", "SILICONFLOW",
    "QWEN", "DASHSCOPE", "GOOGLE", "AZURE", "AWS_", "TOKEN", "SECRET", "PASSWORD", "CREDENTIAL",
    "ACCESS_KEY", "PRIVATE",
];

fn is_secretish(name: &str) -> bool {
    let up = name.to_ascii_uppercase();
    SECRETISH.iter().any(|k| up.contains(k))
}

/// 从当前进程环境里挑出白名单项(且永不含密钥⟦DAMAGE:e7b1⟧⟦DAMAGE:e380⟧pub fn build_env(allow: &[String]) -> Vec<(String, String)> {
    let mut out = Vec::new();
    for key in allow {
        if is_secretish(key) {
            continue; // 白名单写错也不放⟦DAMAGE:e8a1⟧        }
        if let Ok(v) = std::env::var(key) {
            if !v.is_empty() {
                out.push((key.clone(), v));
            }
        }
    }
    out
}

/// 平台解释器。返⟦DAMAGE:e59b⟧(program, [flag, cmd])
pub fn interpreter(cmd: &str) -> (PathBuf, Vec<String>) {
    if cfg!(windows) {
        let com = std::env::var("COMSPEC").unwrap_or_else(|_| "cmd.exe".into());
        (PathBuf::from(com), vec!["/C".into(), cmd.to_string()])
    } else {
        (PathBuf::from("/bin/sh"), vec!["-c".into(), cmd.to_string()])
    }
}

pub struct ShellRunner {

    pub async fn run(&self, args: &Value) -> Result<ShellOutcome, String> {

            .get("command")
            .and_then(Value::as_str)
            .map(str::trim)

            .map_err(|e| format!("arena 目录不可⟦DAMAGE:e794⟧{e}"))?;

            .current_dir(&arena)

fn nonce() -> u64 {
    use std::hash::{Hash, Hasher};
    let mut h = std::collections::hash_map::DefaultHasher::new();
    Instant::now().elapsed().as_nanos().hash(&mut h);
    std::process::id().hash(&mut h);
    h.finish() % 1_000_000
}

/// 头尾各半,中间标记。按字节切但要落在字符边界上⟦DAMAGE:e380⟧pub fn cap_text(s: &str, limit: usize) -> (String, bool) {
    if limit == 0 || s.len() <= limit {
        return (s.to_string(), false);
    }
    let half = limit / 2;
    let head_end = floor_char_boundary(s, half);
    let tail_start = ceil_char_boundary(s, s.len() - half);
    let head = &s[..head_end];
    let tail = &s[tail_start..];
    let dropped = tail_start - head_end;
    (
        format!("{head}\n... truncated {dropped} bytes ...\n{tail}"),
        true,
    )
}

fn floor_char_boundary(s: &str, i: usize) -> usize {
    if i >= s.len() {
        return s.len();
    }
    let mut j = i;
    while j > 0 && !s.is_char_boundary(j) {
        j -= 1;
    }
    j
}

fn ceil_char_boundary(s: &str, i: usize) -> usize {
    if i <= 1 {
        return 0;
    }
    let mut j = i.min(s.len());
    while j < s.len() && !s.is_char_boundary(j) {
        j += 1;
    }
    j
}

/// 杀整个进程树。只杀 shell 本身会留下子进程继续写文件⟦DAMAGE:e380⟧pub fn kill_tree(pid: u32) {
    if pid == 0 {
        return;
    }
    #[cfg(windows)]
    {
        // taskkill /T 杀树。它是系统自带工⟦DAMAGE:e585⟧不需⟦DAMAGE:e8a6⟧unsafe 与额外依赖⟦DAMAGE:e380⟧        let _ = std::process::Command::new("taskkill")
            .args(["/F", "/T", "/PID", &pid.to_string()])
            .stdout(std::process::Stdio::null())
            .stderr(std::process::Stdio::null())
            .spawn();
    }
    #[cfg(unix)]
    {
        // 进程组以 spawn 时的 process_group(0) 建立,pgid == pid
        unsafe {
            libc::killpg(pid as i32, libc::SIGKILL);
        }
    }
}
