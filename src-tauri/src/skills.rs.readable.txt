//! 共享技能库(§9.1):两个 AI 可读、不可写⟦DAMAGE:e380⟧//!
//! 布局:`<workspace>/skills/<slug>/SKILL.md`,frontmatter ⟦DAMAGE:e590⟧name + description⟦DAMAGE:e380⟧//! 系统提示注入**同一份目⟦DAMAGE:e5bd⟧*(name + description 各一⟦DAMAGE:e8a1⟧,`skill_load` 才拿全文⟦DAMAGE:e380⟧//! 因为记忆不共⟦DAMAGE:e4ba⟧"谁先发现哪个技能、谁把它讲给对方"本身就是表演⟦DAMAGE:e380⟧
use serde::{Deserialize, Serialize};
use std::path::Path;

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct SkillMeta {
    pub slug: String,
    pub name: String,
    pub description: String,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct Skill {
    pub meta: SkillMeta,
    pub body: String,
}

/// 极简 frontmatter 解析:`---` 块里⟦DAMAGE:e79a⟧`key: value`⟦DAMAGE:e380⟧fn parse_frontmatter(txt: &str) -> (Vec<(String, String)>, String) {
    let t = txt.strip_prefix('\u{feff}').unwrap_or(txt);
    let mut kv = Vec::new();
    if !t.starts_with("---") {
        return (kv, t.to_string());
    }
    let rest = &t[3..];
    let Some(end) = rest.find("\n---") else {
        return (kv, t.to_string());
    };
    let head = &rest[..end];
    for line in head.lines() {
        if let Some((k, v)) = line.split_once(':') {
            let k = k.trim().to_string();
            let v = v.trim().trim_matches('"').trim_matches('\'').to_string();
            if !k.is_empty() {
                kv.push((k, v));
            }
        }
    }
    let after = &rest[end + 4..];
    (kv, after.trim_start_matches(['\n', '\r']).to_string())
}

/// 扫描技能目⟦DAMAGE:e5bd⟧目录不存在返回空(不报⟦DAMAGE:e994⟧技能是可选增⟦DAMAGE:e5bc⟧⟦DAMAGE:e380⟧pub fn list(root: &Path) -> Vec<SkillMeta> {
    let mut out = Vec::new();
    let Ok(rd) = std::fs::read_dir(root) else {
        return out;
    };
    for entry in rd.flatten() {
        let p = entry.path();
        if !p.is_dir() {
            continue;
        }
        let Some(slug) = p.file_name().map(|s| s.to_string_lossy().to_string()) else {
            continue;
        };
        let f = p.join("SKILL.md");
        let Ok(txt) = std::fs::read_to_string(&f) else {
            continue;
        };
        let (kv, _) = parse_frontmatter(&txt);
        let get = |k: &str| {
            kv.iter()
                .find(|(a, _)| a == k)
                .map(|(_, b)| b.clone())
                .unwrap_or_default()
        };
        let name = if get("name").is_empty() { slug.clone() } else { get("name") };
        out.push(SkillMeta {
            slug,
            name,
            description: get("description"),
        });
    }
    out.sort_by(|a, b| a.slug.cmp(&b.slug));
    out
}

/// ⟦DAMAGE:e68c⟧slug ⟦DAMAGE:e688⟧name 取全文。返⟦DAMAGE:e59b⟧None = 不存在⟦DAMAGE:e380⟧pub fn load(root: &Path, key: &str) -> Option<Skill> {
    let want = key.trim();
    let direct = root.join(want).join("SKILL.md");
    if let Ok(txt) = std::fs::read_to_string(&direct) {
        let (kv, body) = parse_frontmatter(&txt);
        let get = |k: &str| {
            kv.iter()
                .find(|(a, _)| a == k)
                .map(|(_, b)| b.clone())
                .unwrap_or_default()
        };
        return Some(Skill {
            meta: SkillMeta {
                slug: want.to_string(),
                name: if get("name").is_empty() { want.to_string() } else { get("name") },
                description: get("description"),
            },
            body,
        });
    }
    // ⟦DAMAGE:e794⟧name 再找一⟦DAMAGE:e981⟧    list(root)
        .into_iter()
        .find(|m| m.name == want)
        .and_then(|m| load(root, &m.slug))
}

/// 注入系统提示的目录块⟦DAMAGE:e380⟧pub fn catalog_block(metas: &[SkillMeta]) -> String {
    if metas.is_empty() {
        return "共享技能库:当前为空⟦DAMAGE:e380⟧.into();
    }
    let mut s = String::from("共享技能库(只读,双方可用 skill_load 取全⟦DAMAGE:e696⟧:\n");
    for m in metas {
        s.push_str(&format!("- {}(slug:{})—⟦DAMAGE:e280⟧{}\n", m.name, m.slug, m.description));
    }
    s
}

/// 首次建工作区时写入预置技⟦DAMAGE:e883⟧已存在则不覆⟦DAMAGE:e79b⟧⟦DAMAGE:e380⟧pub fn seed_builtin(root: &Path) {
    for (slug, name, desc) in BUILTIN {
        let dir = root.join(slug);
        let f = dir.join("SKILL.md");
        if f.exists() {
            continue;
        }
        let _ = std::fs::create_dir_all(&dir);
        let _ = std::fs::write(
            &f,
            format!("---\nname: {name}\ndescription: {desc}\n---\n\n# {name}\n\n{desc}\n\n用法自由:你们自己决定怎么⟦DAMAGE:e794⟧也可以发明新玩法。\n"),
        );
    }
}
